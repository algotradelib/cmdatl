cmake_minimum_required(VERSION 3.20)
# The release, then the development part (9000: unreleased work after it), as algotradelib and pyatl number theirs.
project(cmdatl VERSION 0.0.0.9000 LANGUAGES CXX)

option(CMDATL_BUILD_TESTS "Build the tests" ON)
option(CMDATL_FETCH_DEPS "Fetch CLI11, toml++ and Catch2 with FetchContent when they are not installed" ON)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# An installed algotradelib: CMAKE_PREFIX_PATH or algotradelib_DIR points at it. Its target brings Boost.JSON.
find_package(algotradelib CONFIG REQUIRED)

include(FetchContent)
find_package(CLI11 2.4 CONFIG QUIET)
if(NOT CLI11_FOUND)
  if(NOT CMDATL_FETCH_DEPS)
    message(FATAL_ERROR "CLI11 2.4 or later is needed: install it, or configure with -DCMDATL_FETCH_DEPS=ON")
  endif()
  FetchContent_Declare(cli11 GIT_REPOSITORY https://github.com/CLIUtils/CLI11.git GIT_TAG v2.4.2 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(cli11)
endif()
# The operator's file is TOML, read by toml++, used header-only whatever its package's own target says: a package
# built as a shared library (Homebrew's defines TOML_HEADER_ONLY=0) throws parse errors whose type does not match
# across the library's boundary, so a refusal would escape as an unknown exception.
find_package(tomlplusplus 3.4 CONFIG QUIET)
if(tomlplusplus_FOUND)
  get_target_property(CMDATL_TOML_INCLUDE tomlplusplus::tomlplusplus INTERFACE_INCLUDE_DIRECTORIES)
else()
  if(NOT CMDATL_FETCH_DEPS)
    message(FATAL_ERROR "toml++ (tomlplusplus) 3.4 or later is needed: install it, or configure with -DCMDATL_FETCH_DEPS=ON")
  endif()
  FetchContent_Declare(tomlplusplus GIT_REPOSITORY https://github.com/marzer/tomlplusplus.git GIT_TAG v3.4.0 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(tomlplusplus)
  set(CMDATL_TOML_INCLUDE ${tomlplusplus_SOURCE_DIR}/include)
endif()
add_library(cmdatl_toml INTERFACE)
target_include_directories(cmdatl_toml SYSTEM INTERFACE ${CMDATL_TOML_INCLUDE})
target_compile_definitions(cmdatl_toml INTERFACE TOML_HEADER_ONLY=1)

# The starters `strategy new` writes, embedded as raw strings so the binary needs no file beside it.
set(CMDATL_TEMPLATE_NAMES mean-reversion crossover momentum breakout)
set(CMDATL_GENERATED ${CMAKE_CURRENT_BINARY_DIR}/generated)
set(cmdatl_entries "")
foreach(name IN LISTS CMDATL_TEMPLATE_NAMES)
  set(file ${CMAKE_CURRENT_SOURCE_DIR}/templates/${name}.atls)
  file(READ ${file} text)
  string(FIND "${text}" ")atls\"" clash)
  if(NOT clash EQUAL -1)
    message(FATAL_ERROR "${file} holds )atls\", which ends its raw string")
  endif()
  string(APPEND cmdatl_entries "  {\"${name}\", R\"atls(${text})atls\"},\n")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${file})
endforeach()
set(file ${CMAKE_CURRENT_SOURCE_DIR}/templates/spec.toml)
file(READ ${file} cmdatl_spec)
string(FIND "${cmdatl_spec}" ")toml\"" clash)
if(NOT clash EQUAL -1)
  message(FATAL_ERROR "${file} holds )toml\", which ends its raw string")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${file})
file(WRITE ${CMDATL_GENERATED}/templates.hpp.new
  "#pragma once\n\n"
  "// Generated by CMake from templates/; edit those files, not this one.\n\n"
  "namespace cmdatl {\n\n"
  "/** @brief A starter program `strategy new` writes. */\n"
  "struct TemplateFile {\n"
  "  /** @brief Its name, for `--template`. */\n"
  "  const char* name_;\n"
  "  /** @brief The program. */\n"
  "  const char* text_;\n"
  "};\n\n"
  "/** @brief The starters, the default first. */\n"
  "inline constexpr TemplateFile TEMPLATES[] = {\n${cmdatl_entries}};\n\n"
  "/** @brief The run spec `strategy new` writes beside a starter; `@NAME@` stands for the program's name. */\n"
  "inline constexpr const char* SPEC_TEMPLATE = R\"toml(${cmdatl_spec})toml\";\n\n"
  "}  // namespace cmdatl\n"
)
configure_file(${CMDATL_GENERATED}/templates.hpp.new ${CMDATL_GENERATED}/cmdatl/templates.hpp COPYONLY)

# The guides `cmdatl guide` prints: the skills' own files, so the guide and the skill are one text.
set(CMDATL_GUIDE_NAMES agent strategy)
set(cmdatl_guides "")
foreach(name IN LISTS CMDATL_GUIDE_NAMES)
  set(file ${CMAKE_CURRENT_SOURCE_DIR}/skills/cmdatl-${name}/SKILL.md)
  file(READ ${file} text)
  string(FIND "${text}" ")md\"" clash)
  if(NOT clash EQUAL -1)
    message(FATAL_ERROR "${file} holds )md\", which ends its raw string")
  endif()
  string(APPEND cmdatl_guides "  {\"${name}\", R\"md(${text})md\"},\n")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${file})
endforeach()
# The run settings' reference, served by `cmdatl strategy reference run-spec`.
set(file ${CMAKE_CURRENT_SOURCE_DIR}/docs/run-spec.md)
file(READ ${file} cmdatl_run_spec)
string(FIND "${cmdatl_run_spec}" ")md\"" clash)
if(NOT clash EQUAL -1)
  message(FATAL_ERROR "${file} holds )md\", which ends its raw string")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${file})
file(WRITE ${CMDATL_GENERATED}/guides.hpp.new
  "#pragma once\n\n"
  "// Generated by CMake from skills/ and docs/run-spec.md; edit those files, not this one.\n\n"
  "namespace cmdatl {\n\n"
  "/** @brief A guide `cmdatl guide` prints: a skill's SKILL.md, whole. */\n"
  "struct GuideFile {\n"
  "  /** @brief Its name, the command's last word. */\n"
  "  const char* name_;\n"
  "  /** @brief The file. */\n"
  "  const char* text_;\n"
  "};\n\n"
  "/** @brief The guides. */\n"
  "inline constexpr GuideFile GUIDES[] = {\n${cmdatl_guides}};\n\n"
  "/** @brief The run settings' reference, docs/run-spec.md, whole. */\n"
  "inline constexpr const char* RUN_SPEC_REFERENCE = R\"md(${cmdatl_run_spec})md\";\n\n"
  "}  // namespace cmdatl\n"
)
configure_file(${CMDATL_GENERATED}/guides.hpp.new ${CMDATL_GENERATED}/cmdatl/guides.hpp COPYONLY)

# The drivers' sources, carried inside cmdatl so `run` and `strategy backtest` build from the same sources as the
# running cmdatl: every header of the modules the run driver may include, its main.cpp and CMakeLists.txt, toml++ with its
# licence, and the backtest driver's main.cpp and CMakeLists.txt.
add_executable(cmdatl_embed tools/embed.cpp)
target_compile_features(cmdatl_embed PRIVATE cxx_std_17)
set(CMDATL_DRIVER_MODULES driver replay orders account system core)
set(cmdatl_driver_pairs "")
set(cmdatl_driver_sources "")
foreach(module IN LISTS CMDATL_DRIVER_MODULES)
  file(GLOB_RECURSE headers CONFIGURE_DEPENDS RELATIVE ${CMAKE_CURRENT_SOURCE_DIR}/include
    ${CMAKE_CURRENT_SOURCE_DIR}/include/cmdatl/${module}/*.hpp)
  foreach(header IN LISTS headers)
    list(APPEND cmdatl_driver_pairs "${header}=${CMAKE_CURRENT_SOURCE_DIR}/include/${header}")
    list(APPEND cmdatl_driver_sources ${CMAKE_CURRENT_SOURCE_DIR}/include/${header})
  endforeach()
endforeach()
# The backtest driver's own two files go under backtest/; `strategy backtest` builds from those, the replay module and
# atlc's output alone.
foreach(file main.cpp CMakeLists.txt backtest/main.cpp backtest/CMakeLists.txt)
  list(APPEND cmdatl_driver_pairs "${file}=${CMAKE_CURRENT_SOURCE_DIR}/driver/${file}")
  list(APPEND cmdatl_driver_sources ${CMAKE_CURRENT_SOURCE_DIR}/driver/${file})
endforeach()
file(GLOB_RECURSE toml_headers CONFIGURE_DEPENDS RELATIVE ${CMDATL_TOML_INCLUDE} ${CMDATL_TOML_INCLUDE}/toml++/*)
foreach(header IN LISTS toml_headers)
  list(APPEND cmdatl_driver_pairs "third_party/${header}=${CMDATL_TOML_INCLUDE}/${header}")
  list(APPEND cmdatl_driver_sources ${CMDATL_TOML_INCLUDE}/${header})
endforeach()
if(EXISTS ${CMDATL_TOML_INCLUDE}/../LICENSE)
  get_filename_component(toml_license ${CMDATL_TOML_INCLUDE}/../LICENSE ABSOLUTE)
  list(APPEND cmdatl_driver_pairs "third_party/toml++/LICENSE=${toml_license}")
  list(APPEND cmdatl_driver_sources ${toml_license})
endif()
add_custom_command(
  OUTPUT ${CMDATL_GENERATED}/cmdatl/driver_files.hpp
  COMMAND cmdatl_embed ${CMDATL_GENERATED}/cmdatl/driver_files.hpp ${cmdatl_driver_pairs}
  DEPENDS cmdatl_embed ${cmdatl_driver_sources}
  COMMENT "Embedding the run driver's sources"
  VERBATIM
)
add_custom_target(cmdatl_driver_files DEPENDS ${CMDATL_GENERATED}/cmdatl/driver_files.hpp)

add_library(cmdatl_core INTERFACE)
target_include_directories(cmdatl_core INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include ${CMDATL_GENERATED})
add_dependencies(cmdatl_core cmdatl_driver_files)
# The package the run driver is built against: the one cmdatl itself was built against, never another.
target_compile_definitions(cmdatl_core INTERFACE CMDATL_ALGOTRADELIB_DIR="${algotradelib_DIR}")
target_link_libraries(cmdatl_core INTERFACE algotradelib::algotradelib CLI11::CLI11 cmdatl_toml)
target_compile_definitions(cmdatl_core INTERFACE CMDATL_VERSION="${PROJECT_VERSION}")
# The credential store `cmdatl auth` keeps logins in: the macOS keychain (no other store in this version).
if(APPLE)
  target_link_libraries(cmdatl_core INTERFACE "-framework Security" "-framework CoreFoundation")
endif()

add_executable(cmdatl src/main.cpp)
target_link_libraries(cmdatl PRIVATE cmdatl_core)
# The library's target names OpenSSL and the brokers' dependencies; cmdatl links only what it uses.
target_link_options(cmdatl PRIVATE
  $<$<PLATFORM_ID:Darwin>:LINKER:-dead_strip_dylibs>
  $<$<PLATFORM_ID:Linux>:LINKER:--as-needed>
)
install(TARGETS cmdatl RUNTIME DESTINATION bin)
# The skills, for a project's .agents/skills/.
include(GNUInstallDirs)
install(DIRECTORY skills/ DESTINATION ${CMAKE_INSTALL_DATADIR}/cmdatl/skills)
# The replay module, for a project's own `main` around `atlc build`'s header; it includes nothing else of cmdatl's.
install(DIRECTORY include/cmdatl/replay/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/cmdatl/replay)

  find_package(Catch2 3 CONFIG QUIET)

    FetchContent_MakeAvailable(catch2)
  endif()
  # A stand-in for atlc whose behaviour each test writes in a script.
  add_executable(fake-atlc tests/fake-atlc.cpp)

    tests/test-cli.cpp tests/test-process.cpp tests/test-strategy.cpp tests/test-acceptance.cpp tests/test-operator.cpp

    tests/test-order-process.cpp tests/test-run-start.cpp tests/test-guides.cpp tests/test-spec.cpp tests/test-auth.cpp

  # forkpty, for the terminal cases of test-auth.
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")

  endif()
  add_dependencies(test-cli cmdatl fake-atlc hold-lock)
  # The documentation checks read README.md and docs/ from the source tree; the process tests run cmdatl itself.
  target_compile_definitions(test-cli PRIVATE
    CMDATL_RUN_TEST_HOOKS
    CMDATL_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}"
    CMDATL_BINARY="$<TARGET_FILE:cmdatl>"
    CMDATL_FAKE_ATLC="$<TARGET_FILE:fake-atlc>"
    CMDATL_HOLD_LOCK="$<TARGET_FILE:hold-lock>"
  )
  add_test(NAME test-cli COMMAND test-cli "~[acceptance]")
  # The acceptance cases compare replays with figures computed in the test: no floating-point contraction on GCC, as
  # the drivers are built.
  target_compile_options(test-cli PRIVATE $<$<CXX_COMPILER_ID:GNU>:-ffp-contract=off>)
  # The store backtest's acceptance case: DuckDB's C library, whose directory CMDATL_TEST_DUCKDB names (a cache
  # variable or the environment), writes its store; without it the case is skipped, and says so.
  if(NOT CMDATL_TEST_DUCKDB AND DEFINED ENV{CMDATL_TEST_DUCKDB})
    set(CMDATL_TEST_DUCKDB "$ENV{CMDATL_TEST_DUCKDB}")
  endif()
  if(CMDATL_TEST_DUCKDB)
    add_executable(make-store tests/make-store.cpp)

    find_library(CMDATL_TEST_DUCKDB_LIBRARY duckdb PATHS ${CMDATL_TEST_DUCKDB} NO_DEFAULT_PATH REQUIRED)
    target_link_libraries(make-store PRIVATE ${CMDATL_TEST_DUCKDB_LIBRARY})
    set_target_properties(make-store PROPERTIES BUILD_RPATH ${CMDATL_TEST_DUCKDB})
    add_dependencies(test-cli make-store)
    target_compile_definitions(test-cli PRIVATE
      CMDATL_TEST_DUCKDB="${CMDATL_TEST_DUCKDB}"
      CMDATL_MAKE_STORE="$<TARGET_FILE:make-store>"
    )
    # The replay module's store loader: its own executable, whose one translation unit defines CMDATL_DUCKDB, so no

    add_executable(test-replay-store tests/test-replay-store.cpp tests/test-store-write.cpp)
    target_link_libraries(test-replay-store PRIVATE algotradelib::algotradelib Catch2::Catch2WithMain ${CMDATL_TEST_DUCKDB_LIBRARY})
    target_include_directories(test-replay-store PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
    target_include_directories(test-replay-store SYSTEM PRIVATE ${CMDATL_TEST_DUCKDB})
    target_compile_definitions(test-replay-store PRIVATE CMDATL_DUCKDB=1)
    target_compile_options(test-replay-store PRIVATE $<$<CXX_COMPILER_ID:GNU>:-ffp-contract=off>)
    set_target_properties(test-replay-store PROPERTIES BUILD_RPATH ${CMDATL_TEST_DUCKDB})
    add_test(NAME test-replay-store COMMAND test-replay-store)
    # The backtest driver that fetches through a broker folder's adapter (here a fixture data vendor), compiled as a
    # backtest's build compiles it.

    target_include_directories(backtest-adapter-vendor PRIVATE

    target_include_directories(backtest-adapter-vendor SYSTEM PRIVATE ${CMDATL_TEST_DUCKDB})
    target_compile_definitions(backtest-adapter-vendor PRIVATE TOML_HEADER_ONLY=1 CMDATL_DUCKDB=1 CMDATL_FETCH_ADAPTER=1)
    target_link_libraries(backtest-adapter-vendor PRIVATE cmdatl_core ${CMDATL_TEST_DUCKDB_LIBRARY})
    set_target_properties(backtest-adapter-vendor PROPERTIES BUILD_RPATH ${CMDATL_TEST_DUCKDB})

  # The replay module with hand-written programs and files written by the test, as the drivers build it: no
  # floating-point contraction on GCC.

  target_compile_options(test-replay PRIVATE $<$<CXX_COMPILER_ID:GNU>:-ffp-contract=off>)
  add_test(NAME test-replay COMMAND test-replay)

    target_compile_definitions(test-faults PRIVATE CMDATL_WRAP_FSYNC)
