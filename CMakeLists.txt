cmake_minimum_required(VERSION 3.20)
# The release, then the development part (9000: unreleased work after it), as algotradelib and pyatl number theirs.
project(cmdatl VERSION 0.0.0.9000 LANGUAGES CXX)

option(CMDATL_BUILD_TESTS "Build the tests" ON)
option(CMDATL_FETCH_DEPS "Fetch CLI11, toml++ and Catch2 with FetchContent when they are not installed" ON)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# An installed algotradelib: CMAKE_PREFIX_PATH or algotradelib_DIR points at it. Its target brings Boost.JSON.
find_package(algotradelib CONFIG REQUIRED)

include(FetchContent)
find_package(CLI11 2.4 CONFIG QUIET)
if(NOT CLI11_FOUND)
  if(NOT CMDATL_FETCH_DEPS)
    message(FATAL_ERROR "CLI11 2.4 or later is needed: install it, or configure with -DCMDATL_FETCH_DEPS=ON")
  endif()
  FetchContent_Declare(cli11 GIT_REPOSITORY https://github.com/CLIUtils/CLI11.git GIT_TAG v2.4.2 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(cli11)
endif()
# The operator's file is TOML, read by toml++, used header-only whatever its package's own target says: a package
# built as a shared library (Homebrew's defines TOML_HEADER_ONLY=0) throws parse errors whose type does not match
# across the library's boundary, so a refusal would escape as an unknown exception.
find_package(tomlplusplus 3.4 CONFIG QUIET)
if(tomlplusplus_FOUND)
  get_target_property(CMDATL_TOML_INCLUDE tomlplusplus::tomlplusplus INTERFACE_INCLUDE_DIRECTORIES)
else()
  if(NOT CMDATL_FETCH_DEPS)
    message(FATAL_ERROR "toml++ (tomlplusplus) 3.4 or later is needed: install it, or configure with -DCMDATL_FETCH_DEPS=ON")
  endif()
  FetchContent_Declare(tomlplusplus GIT_REPOSITORY https://github.com/marzer/tomlplusplus.git GIT_TAG v3.4.0 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(tomlplusplus)
  set(CMDATL_TOML_INCLUDE ${tomlplusplus_SOURCE_DIR}/include)
endif()
add_library(cmdatl_toml INTERFACE)
target_include_directories(cmdatl_toml SYSTEM INTERFACE ${CMDATL_TOML_INCLUDE})
target_compile_definitions(cmdatl_toml INTERFACE TOML_HEADER_ONLY=1)

# The starters `strategy new` writes, embedded as raw strings so the binary needs no file beside it.
set(CMDATL_TEMPLATE_NAMES mean-reversion crossover momentum breakout)
set(CMDATL_GENERATED ${CMAKE_CURRENT_BINARY_DIR}/generated)
set(cmdatl_entries "")
foreach(name IN LISTS CMDATL_TEMPLATE_NAMES)
  set(file ${CMAKE_CURRENT_SOURCE_DIR}/templates/${name}.atls)
  file(READ ${file} text)
  string(FIND "${text}" ")atls\"" clash)
  if(NOT clash EQUAL -1)
    message(FATAL_ERROR "${file} holds )atls\", which ends its raw string")
  endif()
  string(APPEND cmdatl_entries "  {\"${name}\", R\"atls(${text})atls\"},\n")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${file})
endforeach()
set(file ${CMAKE_CURRENT_SOURCE_DIR}/templates/spec.toml)
file(READ ${file} cmdatl_spec)
string(FIND "${cmdatl_spec}" ")toml\"" clash)
if(NOT clash EQUAL -1)
  message(FATAL_ERROR "${file} holds )toml\", which ends its raw string")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${file})
file(WRITE ${CMDATL_GENERATED}/templates.hpp.new
  "#pragma once\n\n"
  "// Generated by CMake from templates/; edit those files, not this one.\n\n"
  "namespace cmdatl {\n\n"
  "/** @brief A starter program `strategy new` writes. */\n"
  "struct TemplateFile {\n"
  "  /** @brief Its name, for `--template`. */\n"
  "  const char* name_;\n"
  "  /** @brief The program. */\n"
  "  const char* text_;\n"
  "};\n\n"
  "/** @brief The starters, the default first. */\n"
  "inline constexpr TemplateFile TEMPLATES[] = {\n${cmdatl_entries}};\n\n"
  "/** @brief The run spec `strategy new` writes beside a starter; `@NAME@` stands for the program's name. */\n"
  "inline constexpr const char* SPEC_TEMPLATE = R\"toml(${cmdatl_spec})toml\";\n\n"
  "}  // namespace cmdatl\n"
)
configure_file(${CMDATL_GENERATED}/templates.hpp.new ${CMDATL_GENERATED}/cmdatl/templates.hpp COPYONLY)

add_library(cmdatl_core INTERFACE)
target_include_directories(cmdatl_core INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include ${CMDATL_GENERATED})
target_link_libraries(cmdatl_core INTERFACE algotradelib::algotradelib CLI11::CLI11 cmdatl_toml)
target_compile_definitions(cmdatl_core INTERFACE CMDATL_VERSION="${PROJECT_VERSION}")

add_executable(cmdatl src/main.cpp)
target_link_libraries(cmdatl PRIVATE cmdatl_core)
# The library's target names OpenSSL and the brokers' dependencies; cmdatl links only what it uses.
target_link_options(cmdatl PRIVATE
  $<$<PLATFORM_ID:Darwin>:LINKER:-dead_strip_dylibs>
  $<$<PLATFORM_ID:Linux>:LINKER:--as-needed>
)
install(TARGETS cmdatl RUNTIME DESTINATION bin)

if(CMDATL_BUILD_TESTS)
  enable_testing()
  find_package(Catch2 3 CONFIG QUIET)
  if(NOT Catch2_FOUND)
    if(NOT CMDATL_FETCH_DEPS)
      message(FATAL_ERROR "Catch2 3 is needed for the tests: install it, or configure with -DCMDATL_FETCH_DEPS=ON")
    endif()
    FetchContent_Declare(catch2 GIT_REPOSITORY https://github.com/catchorg/Catch2.git GIT_TAG v3.8.1 GIT_SHALLOW TRUE)
    FetchContent_MakeAvailable(catch2)
  endif()
  # A stand-in for atlc whose behaviour each test writes in a script.
  add_executable(fake-atlc tests/fake-atlc.cpp)
  add_executable(test-cli
    tests/test-cli.cpp tests/test-process.cpp tests/test-strategy.cpp tests/test-acceptance.cpp
  )
  target_link_libraries(test-cli PRIVATE cmdatl_core Catch2::Catch2WithMain)
  add_dependencies(test-cli cmdatl fake-atlc)
  # The documentation check reads README.md and docs/ from the source tree; the process tests run cmdatl itself.
  target_compile_definitions(test-cli PRIVATE
    CMDATL_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}"
    CMDATL_BINARY="$<TARGET_FILE:cmdatl>"
    CMDATL_FAKE_ATLC="$<TARGET_FILE:fake-atlc>"
  )
  add_test(NAME test-cli COMMAND test-cli "~[acceptance]")
  # Against a real atlc and algotradelib, named by CMDATL_TEST_ATLC and CMDATL_TEST_PREFIX; skipped, and reported
  # as skipped, without them.
  add_test(NAME test-acceptance COMMAND test-cli "[acceptance]")
  set_tests_properties(test-acceptance PROPERTIES SKIP_RETURN_CODE 4 TIMEOUT 3600)
endif()
